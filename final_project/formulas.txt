
Calender Sheet
    Student Selector:
    =IFERROR(VLOOKUP($A$1, {ID2Name!B2:B, ID2Name!A2:A}, 2, FALSE), "")

    Dates formatter: 
    =FILTER(Raw_Reciepts!B18:B, Raw_Reciepts!A18:A = $B$1)

    Calender Generation:
    =ARRAYFORMULA(
    LET(
        base, IFERROR(DATEVALUE($B$2), TODAY()),
        first, DATE(YEAR(base), MONTH(base), 1),
        offset, WEEKDAY(first,2)-1,
        seq, SEQUENCE(6,7,1,1)-1,
        daynum, seq - offset + 1,
        IF((daynum<1)+(daynum>DAY(EOMONTH(first,0))),,
        DATE(YEAR(first), MONTH(first), daynum)
        )
    )
    )

Generated Dates:

    date generators:
        ={"Class Date";
        LET(
        start, DATEVALUE(Important_Dates!A2),
        finish, DATEVALUE(Important_Dates!B2),
        hol, IFERROR(FILTER(Important_Dates!C2:C, Important_Dates!C2:C>=start, Important_Dates!C2:C<=finish), ),
        IF(OR(start="", finish=""), ,
            ARRAYFORMULA(
            WORKDAY.INTL(
                start-1,
                SEQUENCE( NETWORKDAYS.INTL(start, finish, "0000011", hol) ),
                "0000011",
                hol
            )
            )
        )
        )}

Data Analysis:
    ={"UID"; FILTER(ID2Name!A2:A, ID2Name!A2:A<>"")}
    =ARRAYFORMULA({"Names"; IF(A2:A="", "", VLOOKUP(A2:A, ID2Name!A:B, 2, FALSE))}) 
    =ARRAYFORMULA({"Days Missed"; IF(A2:A <> "", TODAY() - Generated_Dates!A2 + 1 - C2:C, "")})
    =ARRAYFORMULA({"Number of Hours"; IF(A2:A <> "", SUMIF(Raw_Reciepts!A:A, A2:A, Raw_Reciepts!E:E), "")})
    =ARRAYFORMULA({"Risk Percentage"; IF(A2:A <> "", 1- (E2:E/(TODAY() - Generated_Dates!A2 + 1)/6), "")})
    =MEDIAN(E2:E)
    =AVERAGE(E2:E)
    =MAX(E2:E)
